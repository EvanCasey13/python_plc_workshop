Variables & Data Types

# Variables
name = "Alice"      # string
age = 20            # integer
gpa = 3.75          # float
active = True       # boolean

# Type conversion
int("42")           # string to int
float("3.14")       # string to float

If Statements

if condition:
    # code
elif another_condition:
    # code
else:
    # code

# Examples
if age >= 18:
    print("Adult")

if score >= 90:
    grade = "A"
elif score >= 80:
    grade = "B"
else:
    grade = "C"

# Combining conditions

    print("OK")

For Loops

# Range
for i in range(5):      # 0,1,2,3,4
    print(i)

for i in range(1, 6):   # 1,2,3,4,5
    print(i)

for i in range(0, 10, 2): # 0,2,4,6,8
    print(i)

# Iterate over list
for item in my_list:
    print(item)

# With index
for i, item in enumerate(my_list):
    print(i, item)

# Loop control
break     # exit loop
continue  # skip to next

While Loops

while condition:
    # code
    # update condition!

# Example: countdown
count = 5
while count > 0:
    print(count)
    count -= 1

# Example: input validation
while True:
    value = input("Enter: ")
    if value == "quit":
        break

Lists

# Creating
nums = [1, 2, 3, 4, 5]
mixed = [1, "two", 3.0]

# Accessing
nums[0]     # first
nums[-1]    # last
nums[1:3]   # slice

# Modifying
nums.append(6)      # add to end
nums.insert(0, 0)   # insert at index
nums.remove(3)      # remove by value

len(nums)           # length
sum(nums)           # sum
max(nums)           # maximum
min(nums)           # minimum
sorted(nums)        # sorted copy
nums.sort()         # sort in place

Dictionaries

# Creating
person = {
    "name": "Alice",
    "age": 20
}

# Accessing
person["name"]
person.get("name")
person.get("email", "N/A")

# Modifying
person["age"] = 21      # update
person["city"] = "NYC"  # add
del person["city"]      # delete

# Looping
for key in person:
    print(key)

for value in person.values():
    print(value)

for key, val in person.items():
    print(key, val)

Functions

# Basic function
def greet():
    print("Hello!")

# With parameters
def greet(name):
    print(f"Hello, {name}!")

# With return value
def add(a, b):
    return a + b

# Default parameters
def greet(name, msg="Hello"):
    print(f"{msg}, {name}!")

# Multiple returns
def stats(nums):
    return min(nums), max(nums)

low, high = stats([1,2,3])

# Variable arguments
def sum_all(*nums):
    return sum(nums)

Common Patterns

# Sum numbers
total = 0
for num in numbers:
    total += num

# Find maximum
max_val = numbers[0]
for num in numbers:
    if num > max_val:
        max_val = num

# Count occurrences
count = 0
for item in items:
    if item == target:
        count += 1

# Build a list
result = []
for item in items:
    result.append(item * 2)

# List comprehension
result = [x * 2 for x in items]

# Filter items
evens = [x for x in nums if x % 2 == 0]

Pixel Art (2D Lists)

# Images are grids of numbers!
picture = [
    [0, 1, 0],
    [1, 1, 1],
    [0, 1, 0],
]

# Draw it
def draw(grid):
    for row in grid:
        line = ""
        for pixel in row:
            if pixel == 1:
                line += "##"
            else:
                line += "  "
        print(line)

draw(picture)
# Output:
#   ##
# ######
#   ##

Simple Chatbot

import random

def reply(message):
    msg = message.lower()

    if "hello" in msg:
        return "Hi there!"

    if "joke" in msg:
        jokes = [
            "Why do programmers...",
            "What's a computer's...",
        ]
        return random.choice(jokes)

    if "bye" in msg:
        return "Goodbye!"

    # Default response
    return "Tell me more!"

# Use it
response = reply("Hello!")
print(response)  # Hi there!
